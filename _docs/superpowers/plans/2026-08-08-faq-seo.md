**Goal:** Portare le FAQ dei 123 articoli con FAQPage a norma con le linee guida Google: testo visibile sempre presente e corrispondente al JSON-LD, reso come menu a tendina con la domanda in heading, domande che intercettano ricerche reali.

**Architecture:** Un solo script Python (`_dev/faq_tool.py`) con quattro sottocomandi che condividono le stesse funzioni di parsing. Nessuna dipendenza esterna: solo stdlib. I file dei post vengono sempre riscritti con `open(path, 'w', encoding='utf-8')`. Una regola CSS accompagna il nuovo markup. Il batch parte solo dopo un pilota verificato in produzione.

**Tech Stack:** Python 3 (stdlib: `re`, `json`, `argparse`, `pathlib`), pytest 9.0.2 per i test, kramdown 2.5.2 come motore di rendering di riferimento, Jekyll 4.3.3 in CI.

## Global Constraints

- Mai usare il tool Edit sui file in `_posts/`: introduce smart quotes che rompono Jekyll. Sempre Python con `open(path, 'w', encoding='utf-8')`.
- Mai em-dash (`—`) nei testi prodotti: usare virgole, punti, parentesi o due punti.
- Il riconoscimento della sezione FAQ usa la regex `^##\s*domande` case-insensitive e ancorata a inizio riga. Mai la stringa esatta "Domande frequenti".
- Il JSON-LD va parsato con `json.loads(..., strict=False)`: alcuni blocchi contengono newline grezzi.
- `sync-visible` è verbatim: nessun contenuto nuovo, il testo esiste già nel JSON-LD.
- Ogni modifica al testo di una domanda o risposta tocca sempre entrambe le copie, JSON-LD e visibile, nella stessa operazione.
- Tutti i contenuti rivolti agli utenti sono in italiano, con accenti corretti.
- Jekyll non è installato in locale: il rendering reale si verifica solo dopo il deploy, aspettando che il run GitHub Actions risulti `completed`.

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Manutenzione delle FAQ dei post: audit, conversione ad accordion, sincronizzazione, verifica."""

import argparse
import json
import re
import sys
from pathlib import Path

POSTS_DIR = Path(__file__).resolve().parent.parent / "_posts"

FAQ_H2 = re.compile(r"^##\s*domande.*$", re.IGNORECASE | re.MULTILINE)
SCRIPT_BLOCK = re.compile(r"<script[^>]*application/ld\+json[^>]*>(.*?)</script>", re.DOTALL | re.IGNORECASE)
BOLD_Q = re.compile(r"^\*\*(.+?)\*\*\s*$", re.MULTILINE)
SUMMARY_Q = re.compile(r"<summary><h3>(.*?)</h3></summary>", re.DOTALL)

INVISIBLE = dict.fromkeys(map(ord, "​‌‍﻿"), None)
QUOTES = {"“": '"', "”": '"', "‘": "'", "’": "'"}


MD_LINK = re.compile(r"\[([^\]]+)\]\([^)]+\)")
MD_EMPHASIS = re.compile(r"(\*{1,2})(.+?)\1")


def normalize(s):
    """Riduce il testo alla sua forma confrontabile.

    Toglie la sintassi markdown (link ed enfasi), uniforma virgolette curve e
    caratteri invisibili. Serve perche' il testo visibile puo' contenere markup
    che nel JSON-LD non c'e', pur essendo lo stesso testo per Google.
    """
    s = MD_LINK.sub(r"\1", s)
    s = MD_EMPHASIS.sub(r"\2", s)
    for bad, good in QUOTES.items():
        s = s.replace(bad, good)
    s = s.translate(INVISIBLE)
    s = re.sub(r"[ \t]+", " ", s)
    return s.strip()


def find_faq_section(text):
    """Restituisce (start, end) della sezione FAQ visibile, o None se assente."""
    m = FAQ_H2.search(text)
    if not m:
        return None
    start = m.start()
    rest = text[start:]
    stop = re.search(r"<script|^##\s(?!.*domande)", rest[len(m.group(0)):], re.IGNORECASE | re.MULTILINE)
    end = start + len(m.group(0)) + (stop.start() if stop else len(rest) - len(m.group(0)))
    return start, end


def parse_jsonld(text):
    """Estrae le coppie domanda/risposta dal blocco FAQPage. None se non c'è."""
    for m in SCRIPT_BLOCK.finditer(text):
        raw = m.group(1).strip()
        if "FAQPage" not in raw:
            continue
        data = json.loads(raw, strict=False)
        pairs = []
        for item in data.get("mainEntity", []):
            pairs.append({
                "q": normalize(item["name"]),
                "a": normalize(item["acceptedAnswer"]["text"]),
            })
        return pairs
    return None


def parse_visible(text):
    """Estrae le coppie domanda/risposta dal testo visibile, in entrambi i formati."""
    span = find_faq_section(text)
    if not span:
        return []
    block = text[span[0]:span[1]]
    if "<summary>" in block:
        return _parse_accordion(block)
    return _parse_bold(block)


def _parse_bold(block):
    pairs = []
    marks = list(BOLD_Q.finditer(block))
    for i, m in enumerate(marks):
        tail = block[m.end():marks[i + 1].start() if i + 1 < len(marks) else len(block)]
        answer = " ".join(p.strip() for p in tail.strip().split("\n") if p.strip())
        pairs.append({"q": normalize(m.group(1)), "a": normalize(answer)})
    return pairs


def _parse_accordion(block):
    pairs = []
    chunks = block.split("<details")
    for chunk in chunks[1:]:
        qm = SUMMARY_Q.search(chunk)
        if not qm:
            continue
        after = chunk[qm.end():].split("</details>")[0]
        answer = " ".join(p.strip() for p in after.strip().split("\n") if p.strip())
        pairs.append({"q": normalize(qm.group(1)), "a": normalize(answer)})
    return pairs
```

```python
def audit(posts_dir=POSTS_DIR):
    """Conta lo stato delle FAQ su tutto il corpus."""
    stats = {"totali": 0, "con_jsonld": 0, "senza_visibile": [], "grassetto": 0,
             "accordion": 0, "domande": 0, "auto_referenziali": [], "em_dash": []}
    for path in sorted(Path(posts_dir).glob("*.md")):
        text = path.read_text(encoding="utf-8")
        stats["totali"] += 1
        pairs = parse_jsonld(text)
        if pairs is None:
            continue
        stats["con_jsonld"] += 1
        span = find_faq_section(text)
        if not span:
            stats["senza_visibile"].append(path.name)
            continue
        block = text[span[0]:span[1]]
        stats["accordion" if "<summary>" in block else "grassetto"] += 1
        for pair in pairs:
            stats["domande"] += 1
            if re.search(r"articolo|nel pezzo|secondo l'autore", pair["q"], re.IGNORECASE):
                stats["auto_referenziali"].append((path.name, pair["q"]))
            if "—" in pair["a"] or "—" in pair["q"]:
                stats["em_dash"].append(path.name)
    return stats


def cmd_audit(args):
    s = audit()
    print(f"post totali:            {s['totali']}")
    print(f"con FAQPage JSON-LD:    {s['con_jsonld']}")
    print(f"  senza testo visibile: {len(s['senza_visibile'])}")
    print(f"  visibile in grassetto:{s['grassetto']}")
    print(f"  visibile in accordion:{s['accordion']}")
    print(f"domande totali:         {s['domande']}")
    print(f"auto-referenziali:      {len(s['auto_referenziali'])}")
    print(f"file con em-dash:       {len(set(s['em_dash']))}")
    return 0


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    sub = parser.add_subparsers(dest="cmd", required=True)
    sub.add_parser("audit", help="stampa lo stato delle FAQ").set_defaults(func=cmd_audit)
    args = parser.parse_args()
    return args.func(args)


if __name__ == "__main__":
    sys.exit(main())
```

```python
def render_accordion(pairs, heading="## Domande frequenti"):
    """Costruisce il blocco FAQ come menu a tendina."""
    out = [heading.rstrip(), ""]
    for pair in pairs:
        out.append('<details class="faq-item" markdown="1">')
        out.append(f'<summary><h3>{pair["q"]}</h3></summary>')
        out.append("")
        out.append(pair["a"])
        out.append("")
        out.append("</details>")
        out.append("")
    return "\n".join(out).rstrip() + "\n"


def to_accordion(text):
    """Converte la sezione FAQ visibile dal formato grassetto al formato accordion."""
    span = find_faq_section(text)
    if not span:
        return text
    block = text[span[0]:span[1]]
    if "<summary>" in block:
        return text
    heading = FAQ_H2.search(text).group(0)
    pairs = _parse_bold(block)
    if not pairs:
        return text
    return text[:span[0]] + render_accordion(pairs, heading) + "\n" + text[span[1]:].lstrip("\n")
```

```python
def compare(text):
    """Confronta JSON-LD e testo visibile. Restituisce la lista dei problemi."""
    jsonld = parse_jsonld(text)
    if jsonld is None:
        return []
    visible = parse_visible(text)
    if not visible:
        return ["manca il testo visibile della FAQPage"]
    if len(jsonld) != len(visible):
        return [f"conteggio diverso: {len(jsonld)} nel JSON-LD, {len(visible)} visibili"]
    problemi = []
    for i, (j, v) in enumerate(zip(jsonld, visible), start=1):
        if j["q"] != v["q"]:
            problemi.append(f"domanda {i} diversa: JSON-LD {j['q']!r} contro visibile {v['q']!r}")
        if j["a"] != v["a"]:
            problemi.append(f"risposta {i} diversa: JSON-LD {j['a'][:60]!r} contro visibile {v['a'][:60]!r}")
    return problemi


def cmd_verify(args):
    rotti = 0
    for path in sorted(POSTS_DIR.glob("*.md")):
        problemi = compare(path.read_text(encoding="utf-8"))
        if problemi:
            rotti += 1
            print(f"\n{path.name}")
            for p in problemi:
                print(f"  - {p}")
    print(f"\nfile non conformi: {rotti}")
```

```python
def sync_visible(text):
    """Rende visibile, verbatim, il testo gia presente nel JSON-LD."""
    pairs = parse_jsonld(text)
    if not pairs:
        return text
    if find_faq_section(text):
        return text
    m = None
    for candidate in SCRIPT_BLOCK.finditer(text):
        if "FAQPage" in candidate.group(1):
            m = candidate
            break
    if not m:
        return text
    blocco = render_accordion(pairs)
    return text[:m.start()].rstrip("\n") + "\n\n" + blocco + "\n" + text[m.start():]


def cmd_sync_visible(args):
    toccati = []
    for path in sorted(POSTS_DIR.glob("*.md")):
        text = path.read_text(encoding="utf-8")
        nuovo = sync_visible(text)
        if nuovo != text:
            if not args.dry_run:
                path.write_text(nuovo, encoding="utf-8")
```
